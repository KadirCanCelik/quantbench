from __future__ import annotations

import heapq
from dataclasses import dataclass, field


@dataclass(order=True)
class PriorityItem:
    priority: int
    item: str = field(compare=False)


class PriorityQueue:
    """A thread-unsafe priority queue wrapper around heapq."""

    def __init__(self) -> None:
        self._elements: list[PriorityItem] = []

    def is_empty(self) -> bool:
        return len(self._elements) == 0

    def put(self, item: str, priority: int) -> None:
        heapq.heappush(self._elements, PriorityItem(priority, item))

    def get(self) -> str:
        if self.is_empty():
            raise IndexError("pop from an empty priority queue")
        return heapq.heappop(self._elements).item
